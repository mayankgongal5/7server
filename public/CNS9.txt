Theory  

A Digital Signature is a cryptographic technique used to verify the
authenticity, integrity, and non-repudiation of a digital message or document.
It is the digital equivalent of a handwritten signature but provides much
stronger security. Digital signatures are created using asymmetric key
cryptography, where the sender signs the message using their private key, and
the receiver verifies the signature using the sender’s public key.  The process
involves generating a hash value of the original message using a hash function.
This hash is then encrypted with the sender’s private key to form the digital
signature. At the receiver’s end, the received message is hashed again and
compared with the decrypted signature hash. If both hashes match, the message is
considered authentic and unaltered. Digital signatures are widely used in secure
emails, software distribution, online transactions, and legal documents.

import java.security.*;


public class DigitalSignatureExample {

	public static void main(String[] args) throws Exception {

		// Generate RSA key pair
		KeyPairGenerator keyGen = KeyPairGenerator.getInstance("RSA");
		keyGen.initialize(1024);
		KeyPair pair = keyGen.generateKeyPair();

		PrivateKey privateKey = pair.getPrivate();
		PublicKey publicKey = pair.getPublic();

		String message = "Save me";

		// Create Signature object
		Signature signature = Signature.getInstance("SHA256withRSA");

		// Signing
		signature.initSign(privateKey);
		signature.update(message.getBytes());
		byte[] digitalSignature = signature.sign();

		System.out.println("Digital Signature Generated");

		// Verification
		signature.initVerify(publicKey);
		signature.update(message.getBytes());
		boolean isVerified = signature.verify(digitalSignature);

		System.out.println("Signature Verified: " + isVerified);
	}
}
